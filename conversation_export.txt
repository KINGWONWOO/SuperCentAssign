### ✅ 업그레이드 cost UI 수정 (`UpgradeZone.cs`)

`AttachDrill()`/`DetachDrill()` 제거. 드릴 표시/숨김은 `PlayerToolManager.toolModels[1]`이 단독 관리 (구 코드는 씬 오브젝트와 별도 instantiate가 동시에 발생해서 두 개가 겹쳤음)

`pickaxeModel` → `ToolModel_Pickaxe` 씬 연결 완료. `Start()`에서 `mixamorig:RightHand` 본에 자동 부착. **Inspector에서 오프셋 조정 가능**:

### ✅ 돌 파괴 사운드 + 파티클 인프라 (`RockNode.cs`)

### 데이터 흐름

| 완료 | `ExecuteUpgrade()` → `costText = "DONE!"` | `DONE!` |

### 실제 함수들

```csharp

     public class MiningAnimNotifier : MonoBehaviour
     {
         public System.Action OnMiningImpact;
         private Animator animator;
         private static readonly int IsMiningHash = Animator.StringToHash("IsMining");
         private bool fired = false;
         // 1.267s clip: 0.85=1.077s, 0.2s earlier → 0.877s/1.267s ≈ 0.69
         private const float Threshold = 0.69f;
         void Awake() => animator = GetComponent<Animator>();
         void Update()
         {
             if (animator == null || !animator.GetBool(IsMiningHash)) { fired = false; return; }
             if (animator.IsInTransition(0)) return;
             float norm = animator.GetCurrentAnimatorStateInfo(0).normalizedTime % 1f;
             if (!fired && norm >= Threshold) { fired = true; OnMiningImpact?.Invoke(); }
             else if (norm < Threshold * 0.4f) { fired = false; }
         }
     }
     ```
