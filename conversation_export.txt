     ```csharp
     static void AddFBXVisual(GameObject parent, string fbxPath, string matName,
                              Vector3 targetBounds, Vector3 localOffset, bool uniform,
                              bool preserveFBXRot = true)
     {
         var fbx = AssetDatabase.LoadAssetAtPath<GameObject>(fbxPath);
         if (fbx == null) { Debug.LogWarning("[FBXDesigner] FBX not found: " + fbxPath); return; }
         var existingVis = parent.transform.Find("Visual");
         if (existingVis != null) Object.DestroyImmediate(existingVis.gameObject);
         var vis = (GameObject)Object.Instantiate(fbx, parent.transform);
         vis.name = "Visual";
         vis.transform.localPosition = localOffset;
         if (!preserveFBXRot)
             vis.transform.localRotation = Quaternion.identity;
         var mat = GetMat(matName);
         if (mat != null)
             foreach (var r in vis.GetComponentsInChildren<MeshRenderer>())
                 r.sharedMaterial = mat;
         if (preserveFBXRot)
         {
             if (uniform) ScaleUniformBaked(vis, targetBounds);
             else ScaleNonUniformBaked(vis, targetBounds);
         }
         else
         {
             if (uniform) ScaleNonUniform(vis, targetBounds);
             else ScaleNonUniform(vis, targetBounds);
         }
     }
     ```
     - Current `ScaleNonUniformBaked`:
     ```csharp
     // FBX baked 270°X (Z-up) 회전용 비균등 스케일.
     // Unity: world X = local X,  world Y = local Z,  world Z = local Y
     // → b.size.x = naturalLocalX, b.size.y = naturalLocalZ, b.size.z = naturalLocalY
     // localScale: (tW.x/b.x, tW.z/b.z, tW.y/b.y)
     static void ScaleNonUniformBaked(GameObject go, Vector3 targetWorldSize)
     {
         go.transform.localScale = Vector3.one;
         var renderers = go.GetComponentsInChildren<Renderer>();
         if (renderers.Length == 0) return;
         var b = renderers[0].bounds;
         foreach (var r in renderers) b.Encapsulate(r.bounds);
         if (b.size.x < 0.001f || b.size.y < 0.001f || b.size.z < 0.001f) return;
         go.transform.localScale = new Vector3(
             targetWorldSize.x / b.size.x,
             targetWorldSize.z / b.size.z,   // worldZ target → local Y scale
             targetWorldSize.y / b.size.y);  // worldY target → local Z scale
     }
     ```
     - Current `ScaleByHeight`:
     ```csharp
     static void ScaleByHeight(GameObject go, float targetHeight)
     {
         go.transform.localScale = Vector3.one;
         var renderers = go.GetComponentsInChildren<Renderer>();
         if (renderers.Length == 0) return;
         var b = renderers[0].bounds;
         foreach (var r in renderers) b.Encapsulate(r.bounds);
         if (b.size.y < 0.001f) return;
         go.transform.localScale = Vector3.one * (targetHeight / b.size.y);
     }
     ```
     - Current `ScaleFenceSegment`:
     ```csharp
     static void ScaleFenceSegment(GameObject go, Vector3 wallLocalSize)
     {
         go.transform.localScale = Vector3.one;
         var renderers = go.GetComponentsInChildren<Renderer>();
         if (renderers.Length == 0) return;
         var b = renderers[0].bounds;
         foreach (var r in renderers) b.Encapsulate(r.bounds);
         if (b.size.x < 0.001f || b.size.y < 0.001f) return;
         float wallLength = Mathf.Max(wallLocalSize.x, wallLocalSize.z);
         float wallHeight = Mathf.Max(wallLocalSize.y, 1.0f);
         // baked: world X = fence width, world Y = fence height
         float sx = (wallLength > 0.1f) ? wallLength / b.size.x : 1.0f;
         float sy_world = wallHeight / b.size.y;
         // localScale Z → world Y (baked 270°X)
         go.transform.localScale = new Vector3(sx, 1.0f, sy_world);
     }
     ```

     ```csharp
         if (!stackManager.HasItemOfType(ItemType.Ore)) return;
         if (converterMachine == null) return;
         StackItem ore = stackManager.RemoveTopItemOfType(ItemType.Ore);
         if (ore == null) return;

     }
     private System.Collections.IEnumerator OreDropAndFly(GameObject oreObj)
     {
         PushItem(oreObj);
         yield return new WaitForSeconds(0.5f);
         if (oreObj == null) yield break;
         items.Remove(oreObj);
         Vector3 startPos = oreObj.transform.position;
         oreObj.transform.SetParent(null);
         oreObj.transform.position = startPos;
         Vector3 endPos = converterMachine.transform.position + Vector3.up * 0.5f;
         float duration = 0.35f; float elapsed = 0f;
         while (elapsed < duration && oreObj != null) {
             elapsed += Time.deltaTime;
             float t = Mathf.Clamp01(elapsed / duration);
             Vector3 pos = Vector3.Lerp(startPos, endPos, t);
             pos.y += Mathf.Sin(t * Mathf.PI) * 0.8f;
             oreObj.transform.position = pos;
             yield return null;
         }
         if (oreObj != null) { converterMachine.ReceiveOre(1); Destroy(oreObj); }
         for (int i = 0; i < items.Count; i++)
             if (items[i] != null)
                 items[i].transform.localPosition = new Vector3(0f, i * stackSpacingY, 0f);
     }
     ```
